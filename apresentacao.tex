    \section{Introdução}
	\begin{frame}{Introdução}{Contextualização da Rede Sem Fio Universitária}	
        \begin{itemize}
            \item As redes locais sem fio (\textit{Wireless Local Area Networks} -- WLANs) corporativas e universitárias lidam com alta densidade de usuários e variabilidade de perfis de tráfego.
            \item No campus do Instituto de Ciências Exatas e Aplicadas (ICEA/UFOP), a infraestrutura atende estudantes, professores e servidores com demandas diversificadas.
            \item Cenário pós-pandemia: Aumento drástico do consumo de banda por aplicações síncronas de vídeo, chamadas remotas e plataformas de apoio pedagógico.
            \item Infraestrutura híbrida sob análise: Controladoras e pontos de acesso de diferentes fabricantes concorrendo no mesmo meio físico de RF (Ruckus e UniFi) \cite{cisco2020design}.
        \end{itemize} 
    \end{frame}

        Ponto-chave: coexistem dois fabricantes sem coordenação de RF no mesmo espaço físico. \\
        Ruckus SmartZone = controladora centralizada de alta performance. \\
        UniFi = rede mais simples, sem coordenação ativa de canal.

    \section{Motivação e Objetivos}
    \begin{frame}{Motivação}
        \begin{itemize}
            \item \textbf{Desafio da Alta Densidade}: Práticas de radiofrequência mostram que muitos clientes ativos degradam severamente o tempo aéreo comum (\textit{airtime}) \cite{gast2005}.
            \item \textbf{Obsolescência e Heterogeneidade}: Equipamentos corporativos de ponta (Ruckus R600) coexistindo com redes mais simples (UniFi) sem coordenação de RF.
            \item \textbf{Percepção do Usuário}: Frequente insatisfação com a estabilidade e velocidade da conexão de rede corporativa, necessitando de um diagnóstico livre de viés qualitativo puramente opinativo.
            \item \textbf{Ociosidade vs. Abuso}: Falta de visibilidade técnica sobre o uso abusivo de banda por poucos usuários limitando a estabilidade geral dos demais.
        \end{itemize}
    \end{frame}
    \note{
        [~1,5 min] CSMA/CA: o Wi-Fi é um meio compartilhado. Cada dispositivo ativo reduz a fatia de airtime dos outros. \\
        Ruckus R600 usa BeamFlex (direcionamento dinâmico de feixe) --- vantagem física real sobre antenas omnidirecionais. \\
        Ponto político: a insatisfação dos alunos é real, mas nunca tinha sido quantificada com dados de produção. Esse trabalho faz isso.
    }
    
    \begin{frame}{Objetivos}
        \begin{block}{Objetivo Geral}
            Caracterizar e diagnosticar os gargalos físicos e lógicos que limitam o desempenho e a qualidade de experiência nas redes sem fio corporativas do ICEA/UFOP.
        \end{block}
        \begin{block}{Objetivos Específicos}
            \begin{itemize}
                \item Coleta automatizada de dados quantitativos de logs de controladoras (SmartZone e UniFi Controller) e logs RADIUS no firewall pfSense.
                \item Pré-processamento, limpeza, unificação e padronização das métricas.
                \item Modelagem heurística determinística de classificação de gargalos.
                \item Aplicação de agrupamento não-supervisionado (\textit{K-Means}) para validação.
                \item Elaboração de sugestões e recomendações práticas de QoS e gerenciamento de RF.
            \end{itemize}
        \end{block}
    \end{frame}
    \note{
        [~1 min] Objetivo GERAL: caracterizar e diagnosticar --- não apenas descrever, mas propor soluções. \\
        Específicos: a sequência lógica vai de coleta → padronização → classificação heurística → validação K-Means → recomendações. \\
        Se a banca perguntar por que usar K-Means: porque valida os limiares heurísticos SEM supervisão, eliminando viés manual.
    }

    \begin{frame}{Metodologia}{Fluxo de Processamento e Limpeza}
        \begin{enumerate}
            \item \textbf{Limpeza e Deduplicação}: Eliminação de logs vazios, duplicados e inconsistências temporais.
            \item \textbf{Tratamento de Nulos}: Tratamento de variáveis como o \textit{Link Speed} do Ruckus e o \textit{Client Connection Quality} (\textit{CCQ}) da UniFi.
            \item \textbf{Padronização Métricas de RF}:
                \begin{itemize}
                    \item Intensidade de Sinal (IS) e Ruído de Fundo mapeados em \textbf{dBm}.
                    \item RSSI / SNR padronizados em \textbf{dB}.
                    \item Throughput total, download (TX) e upload (RX) em \textbf{Mbps}.
                    \item Volume total trafegado em \textbf{MB}.
                \end{itemize}
        \end{enumerate}
    \end{frame}
    \note{
        [~1,5 min] Unificação crítica: Ruckus reporta throughput em Kbps, UniFi em Mbps separado TX/RX --- tudo normalizado para Mbps total. \\
        CCQ (Client Connection Quality) da UniFi: métrica proprietária de 0 a 100 que combina retransmissão e força de sinal --- usada como proxy de qualidade. \\
        IS (Intensidade de Sinal) em dBm: lembrar que valores MAIS NEGATIVOS = sinal mais fraco (ex: -90 dBm é péssimo, -50 dBm é ótimo).
    }

    \begin{frame}{Metodologia}{Limiares Determinísticos de Gargalos}
        Construção de uma árvore de decisão baseada em 5 perfis heurísticos de gargalos de rede:
        \begin{itemize}
            \item \textbf{G1: Baixa Qualidade de Enlace}: Intensidade de Sinal ($IS < -75$ dBm).
            \item \textbf{G2: Congestionamento do Meio}: Densidade do AP ($\ge 15$ clientes simultâneos por AP/minuto).
            \item \textbf{G3: Interferência ou SNR Ruim}: Ruído permanente ($Noise > -90$ dBm) ou relação sinal-ruído marginal ($SNR < 20$ dB) mesmo com boa IS ($\ge -75$ dBm).
            \item \textbf{G4: Instabilidade de Roaming}: Dispositivos instáveis ($> 5$ transições de AP/dia).
            \item \textbf{G5: Gargalo Cabeado (Fast Ethernet)}: Saturação da interface uplink do AP ($\ge 90$ Mbps).
        \end{itemize}
    \end{frame}
    \note{
        [~2 min] Explicar cada gargalo com calma --- a banca pode perguntar sobre os limiares. \\
        G1 (-75 dBm): limiar padrão IEEE 802.11 para cobertura mínima aceitável. Abaixo disso, modulação cai para MCS0 (BPSK) --- velocidade mínima. \\
        G2 (15 clientes/AP): baseado em práticas de engenharia de RF corporativa (Cisco e Ruckus Design Guides). \\
        G3 (SNR < 20 dB): margem mínima para operação confiável em 802.11n/ac. \\
        G4 (5 transições/dia): roaming excessivo degrada sessões TCP e VoIP. \\
        G5 (90 Mbps): 90\% da capacidade nominal de 100 Mbps da interface Fast Ethernet --- gargalo de uplink físico cabeado.
    }

    \begin{frame}{Metodologia}{Agrupamento Operacional (K-Means)}
        Para validação dos gargalos de forma não-supervisionada, estruturou-se o algoritmo K-Means.
        \begin{block}{Vetor de Features Euclidiano}
            $$X_j = \{ C_j, T_j, S_j, R_j \}$$
            \begin{itemize}
                \item $C_j$: Número médio de clientes conectados.
                \item $T_j$: Throughput total médio do AP (Mbps).
                \item $S_j$: Intensidade de Sinal física recebida média (IS em dBm).
                \item $R_j$: Taxa média de retransmissão de quadros (\%).
            \end{itemize}
        \end{block}
        \begin{itemize}
            \item Variáveis normalizadas via \textit{StandardScaler} ($z$-score).
            \item Número ótimo de clusters definido em $K=4$ por análise matemática conjunta do Método do Cotovelo, Coeficiente de Silhueta e Índice Davies-Bouldin.
        \end{itemize}
    \end{frame}
    \note{
        [~1,5 min] Vetor de 4 features por AP: clientes médios (C), throughput médio (T), IS média (S), retransmissão média (R). \\
        StandardScaler: transforma cada variável em z-score (média 0, desvio 1) para evitar que throughput (em Mbps) domine a distância euclidiana sobre IS (em dBm). \\
        K=4 escolhido por consenso de três critérios: cotovelo (inércia), silhueta (coesão/separação) e Davies-Bouldin (compactação intra-cluster). \\
        Se a banca perguntar por que K-Means e não DBSCAN ou hierárquico: K-Means é interpretável e eficiente para o tamanho do dataset.
    }

    \section{Recomendações e QoS}
    \begin{frame}{Recomendações e Ações Práticas}
        Com base no diagnóstico dos gargalos físicos e perfis de APs, sugerimos as seguintes intervenções de engenharia de rede:
        
        \begin{block}{Solução para Gargalo G1 (Baixa Qualidade de Enlace)}
            Remanejamento físico de APs do Grupo D para reduzir barreiras físicas de atenuação de RF e preencher vales de cobertura.
        \end{block}
        
        \begin{block}{Solução para Gargalos G2 e G3 (Congestionamento e Interferência)}
            \begin{itemize}
                \item Ativação do \textbf{Band Steering} ativo em ambas as redes para direcionar clientes com rádio compatível de 2,4 GHz para a banda de 5 GHz.
                \item Ajuste dinâmico de canais de RF estáticos e não-sobrepostos nas controladoras UniFi e redução do limiar de potência máxima de transmissão do rádio para mitigar a assimetria de enlace do Grupo C \cite{santos2017qos}.
            \end{itemize}
        \end{block}
    \end{frame}
    \note{
        [~1,5 min] G1 (Grupo D Ruckus): ação de custo baixo --- revisar posicionamento físico dos APs com atenuação de parede severa. \\
        Band Steering: configuração de software nas controladoras, ZERO custo adicional de hardware. \\
        Redução de potência UniFi: contra-intuitivo, mas reduzir potência melhora a assimetria de enlace --- o cliente passa a ouvir menos APs distantes. \\
        Potencial melhora: se Grupo C UniFi (39\% retransmissão) for resolvido, a QoE de todos os usuários conectados naqueles APs melhora drasticamente.
    }

    \section{Conclusões}
    \begin{frame}{Conclusões}
        \begin{itemize}
            \item O estudo coletou e caracterizou um dataset de \textbf{498.034 registros de telemetria reais} no campus (20 dias com registros de log), fornecendo um diagnóstico preciso de rede de produção.
            \item Validou-se que a insatisfação com a rede sem fio corporativa decorre de contenção de meio e assimetria de potência física (Gargalos G2 e G3), e não por limites de banda cabeada (Gargalo G5).
            \item A modelagem K-Means foi eficiente em agrupar access points de acordo com a saúde operacional da célula sem intervenção de julgamentos manuais.
            \item Ações sugeridas de gerência de rádio (potência de rádio e Band Steering) representam intervenções de baixo custo e alto impacto prático na Qualidade de Experiência (QoE) dos usuários.
        \end{itemize}
