    \section{Introdução}
	\begin{frame}{Introdução}{Cenário do Campus e Infraestrutura}

            \item \textbf{Conexão Externa}: Link dedicado corporativo de \textbf{1 Gbps} (RNP).
            \item \textbf{Parque de Pontos de Acesso (43 APs catalogados)}:
            \begin{itemize}
                \item \textbf{24 APs Ruckus R600}: Operação \textit{Dual-band} (salas de aula, laboratórios e biblioteca).
                \item \textbf{19 APs UniFi}: Operação em 2,4 GHz (gabinetes docentes e áreas administrativas).
            \end{itemize}
            \item \textbf{Demanda Concorrente}: Picos superiores a \textbf{600 clientes simultâneos} disputando o meio aéreo.
        \end{itemize}

    \begin{frame}{Introdução}{O Problema de Pesquisa}	
        \begin{itemize}
            \item \textbf{Percepção vs. Realidade}: Relatos frequentes de lentidão atribuídos genericamente ao ``link de internet'', sem distinção das restrições físicas da interface sem fio.
            \item \textbf{Falta de Visibilidade Unificada}: Sistemas heterogêneos de gerência (SmartZone e UniFi) sem métricas consolidadas de desempenho e RF.
            \item \textbf{Questão Central}: \textit{Onde residem os gargalos reais da WLAN no campus (sinal, concorrência, interferência, mobilidade ou cabeamento) e como diagnosticá-los empiricamente?}
        \end{itemize}
	\end{frame}
    \note{
        [~1 min] Este slide apresenta o cerne do problema prático e científico: \\
        A insatisfação dos usuários com a rede sem fio é recorrente, mas quase sempre rotulada genericamente como 'problema na internet'. \\
        A questão investigada foi: o gargalo decorre do link da RNP, de limitações em portas de switches ou da física da interface de radiofrequência? \\
        Sem dados consolidados de telemetria entre Ruckus e UniFi, a tomada de decisão em infraestrutura torna-se empírica no sentido leigo. O trabalho propõe uma metodologia quantitativa para mapear os gargalos operacionais da rede.

    \section{Motivação e Objetivos}
    \begin{frame}{Motivação}
        \begin{itemize}
            \item \textbf{Contenção no Meio Aéreo}: No protocolo CSMA/CA, múltiplos clientes dividem o tempo de antena (\textit{airtime}), elevando colisões e latência \cite{gast2005}.
            \item \textbf{Ambiente Heterogêneo}: Coexistência de plataformas distintas (Ruckus e UniFi) sem coordenação dinâmica de rádio.
            \item \textbf{Assimetria de Demanda}: Tráfego concentrado em poucos dispositivos reduz a disponibilidade do canal para os demais.
            \item \textbf{Diagnóstico Baseado em Dados}: Substituir percepções subjetivas por telemetria contínua coletada em produção.
        \end{itemize}
    \end{frame}
    \note{
        [~1,5 min] CSMA/CA: o Wi-Fi é um meio compartilhado por contenção. Cada estação ativa consome uma fração de tempo de antena (airtime). \\
        A Ruckus R600 dispõe de antenas adaptativas e opera em dual-band (2,4 e 5 GHz), enquanto a UniFi legada atua em 2,4 GHz em canal único. \\
        A motivação central é fornecer dados técnicos de produção que permitam diagnosticar a origem real das limitações observadas pelos usuários.
    }
    
    \begin{frame}{Objetivos}
        \begin{block}{Objetivo Geral}
            Caracterizar e diagnosticar gargalos de desempenho na WLAN corporativa do campus ICEA/UFOP a partir de dados telemétricos.
        \end{block}
        \begin{block}{Objetivos Específicos}
            \begin{itemize}
                \item Coletar, tratar e unificar dados de produção das controladoras SmartZone (SNMP) e UniFi (REST API).
                \item Avaliar correlações e contrastes estatísticos entre grandezas de RF.
                \item Classificar gargalos operacionais por meio de modelo heurístico (G1 a G5).
                \item Identificar perfis naturais de APs via agrupamento não-supervisionado (\textit{K-Means}).
                \item Propor ações práticas de otimização de infraestrutura e RF de baixo custo.
            \end{itemize}
        \end{block}
    \end{frame}
    \note{
        [~1 min] Objetivo Geral: caracterizar e diagnosticar as limitações operacionais da rede sem fio do campus. \\
        Específicos: a sequência metodológica parte da ingestão de telemetria, passa por tratamento e classificação determinística (G1 a G5), agrega análise não-supervisionada via K-Means como evidência complementar de agrupamento dos APs, e culmina em intervenções práticas de rede.
    }

    \begin{frame}{Metodologia}{Limiares Operacionais de Gargalos}
        Critérios heurísticos para caracterização dos estados da rede:
        \begin{itemize}
            \item \textbf{G1: Sinal Atenuado}: Sinal fraco ($IS < -75$ dBm), reduzindo a velocidade de modulação \cite{cisco2020design}.
            \item \textbf{G2: Contenção de Meio}: Disputa no mesmo rádio ($\ge 15$ clientes simultâneos) \cite{gast2005}. Picos observados atingiram até \textbf{64 clientes/AP}.
            \item \textbf{G3: Interferência / Ruído}: Ruído elevado no canal ($Noise > -90$ dBm ou SNR $< 20$ dB).
            \item \textbf{G4: Instabilidade de Roaming}: Troca excessiva de antenas ($> 15$ vezes ao dia), gerando oscilações e quedas momentâneas.
            \item \textbf{G5: Saturação do Cabo}: Tráfego no teto da porta cabeada ($\ge 90$ Mbps em portas Fast Ethernet de 100 Mbps).
        \end{itemize}
    \end{frame}
    \note{
        [~1,5 min] Explicar a lógica prática dos 5 gargalos: \\
        1. G1 (Sinal < -75 dBm): sinal atenuado pela distância ou obstáculos, forçando o rádio a transmitir em velocidades mais baixas. \\
        2. G2 (>= 15 clientes): no Wi-Fi apenas um fala por vez. Com 15 ou mais clientes disputando o ar, a espera e as colisões disparam. No campus vimos picos de mais de 60 clientes em um único AP. \\
        3. G3 (Ruído > -90 dBm): interferência externa ou canal poluído por ruído. \\
        4. G4 (> 15 trocas de antena/dia): identifica dispositivos que ficam pulando de antena em antena ou alternando em falso, provocando quedas e reautenticações frequentes. \\
        5. G5 (>= 90 Mbps): verifica se a lentidão vinha do cabo dos switches antigos de 100 Mbps em vez do rádio.
    }

        \textbf{[Tempo: ~1,5 min | Roteiro de Fala - Correlações de Spearman]}\\[2pt]
        $\bullet$ \textbf{Por que Spearman em vez de Pearson?} O teste de Shapiro-Wilk rejeitou a normalidade. Spearman avalia associações monotônicas (não-lineares), sendo robusto contra caudas pesadas e leituras atípicas.\\[2pt]
        $\bullet$ \textbf{1. Sinal x Throughput (0,03 RK vs 0,46 UF - Achado Contraintuitivo):} Na Ruckus, a correlação é quase nula ($r_s = 0,03$). A intuição comum sugeriria que sinal forte proporcionaria maior velocidade. No entanto, no protocolo CSMA/CA, o meio é compartilhado e a grande maioria dos clientes está ociosa, desacoplando o sinal da vazão instantânea. Na UniFi ($r_s = 0,46$), o rádio legado em 2,4 GHz faz com que a atenuação restrinja a modulação básica (MCS) dos clientes distantes.\\[2pt]
        $\bullet$ \textbf{2. Sinal x Retransmissões (0,17 RK vs -0,34 UF):} Na UniFi, correlação negativa (-0,34): sinal fraco eleva a perda física de pacotes. Na Ruckus (+0,17), o enlace fraco é sustentado pelas antenas adaptativas, mas estações próximas em salas cheias sofrem colisões aéreas.\\[2pt]
        $\bullet$ \textbf{3. Ruído x Retransmissões (-0,64 RK vs 0,02 UF):} Frise à banca: trata-se de um resultado inesperado que indica a necessidade de interpretar com cautela a métrica de ruído fornecida pela controladora, influenciada pela sensibilidade dinâmica da plataforma (evitando especulações físicas prematuras).\\[2pt]
        $\bullet$ \textbf{4. Volume x Tempo de Sessão (0,70 RK vs 0,36 UF):} Usuários na Ruckus apresentam conexões mais estáveis e prolongadas durante as aulas.\\[2pt]
        $\bullet$ \textbf{Nota de Densidade:} Agrupando dados por rádio, a presença de clientes eleva a vazão somada do AP ($r_s = 0,80$ RK / $0,56$ UF).

    \begin{frame}[label=slide_testes]{Resultados}{Inferência Estatística e Equidade}
        \begin{columns}
            \begin{column}{0.50\textwidth}
                \scriptsize
                \begin{itemize}
                    \item \textbf{Não-Normalidade (Shapiro-Wilk)}: Rejeição de normalidade ($p < 0,001$) em todas as métricas, justificando testes não-paramétricos.
                    \item \textbf{Comparação entre Redes (Mann-Whitney U)}: Diferenças estatisticamente significativas ($p < 0,001$) em sinal, vazão e retransmissões.
                    \item \textbf{Tamanho de Efeito ($r$ de Rosenthal)}:
                    \begin{itemize}
                        \scriptsize

            - \textbf{O que significa o Eixo Y (Densidade)?} O eixo Y representa a estimativa de densidade de probabilidade (KDE), refletindo a \textbf{frequência temporal relativa} com que a rede operou em cada nível de equidade ao longo dos 49 dias. A área total sob a curva é igual a 1 (100\% do tempo).\\[2pt]
            - \textbf{Interpretação dos Picos:} A curva da UniFi atinge seu pico em $0,13$, e a Ruckus em $0,26$. Isso reflete a forte assimetria instantânea gerada pela dinâmica do CSMA/CA e pela concentração de alunos em salas de aula específicas (Gargalo G2).\\[2pt]
            - \textbf{Densidade em 1,0:} Próximo a $1,0$ (equilíbrio instantâneo estrito), a densidade é praticamente nula, o que é esperado em redes móveis onde as demandas dos clientes ocorrem em rajadas assíncronas.\\[2pt]
            - \textbf{Piso Teórico de Jain:} Varia de $1/N = 1/43 \approx 0,023$ (onde 1 rádio concentra todo o tráfego e 42 ficam ociosos) até $1,0$ (partilha idêntica).
